cmake_minimum_required(VERSION 3.20)
project(pvz_a11y LANGUAGES CXX)
set(TAO_NATIVE_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Tao Native SDK root")
set(PRISM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/prism-bin/windows-x64" CACHE PATH "Prism prebuilt runtime root (include/, dynamic/release/bin/)")
set(TAO_PLATFORM_SDK "${TAO_NATIVE_SDK}")
if(ANDROID AND EXISTS "${TAO_NATIVE_SDK}/platforms/android_arm64")
    set(TAO_PLATFORM_SDK "${TAO_NATIVE_SDK}/platforms/android_arm64")
elseif(WIN32 AND EXISTS "${TAO_NATIVE_SDK}/platforms/windows_x64")
    set(TAO_PLATFORM_SDK "${TAO_NATIVE_SDK}/platforms/windows_x64")
endif()

add_library(pvz_a11y SHARED src/Plugin.cpp src/ChallengeScreenState.cpp src/MainMenuState.cpp src/SeedChooserState.cpp)
target_include_directories(pvz_a11y PRIVATE src "${TAO_NATIVE_SDK}/include" "${TAO_PLATFORM_SDK}/include")
target_include_directories(pvz_a11y PRIVATE "${TAO_PLATFORM_SDK}/include/engine" "${TAO_PLATFORM_SDK}/include/engine/SexyAppFramework")
if(WIN32)
    if(EXISTS "${PRISM_ROOT}/include/prism.h")
        target_compile_definitions(pvz_a11y PRIVATE PVZ_A11Y_WITH_PRISM)
        target_include_directories(pvz_a11y PRIVATE "${PRISM_ROOT}/include")
        foreach(dll prism.dll tolk.dll)
            configure_file("${PRISM_ROOT}/dynamic/release/bin/${dll}" "${CMAKE_CURRENT_BINARY_DIR}/package/native/windows_x64/${dll}" COPYONLY)
        endforeach()
    else()
        message(WARNING "PRISM_ROOT not found, building without prism support (SAPI only)")
    endif()
    include("${TAO_NATIVE_SDK}/cmake/TaoNativePlugin.cmake")
    tao_configure_native_plugin(pvz_a11y "${CMAKE_CURRENT_BINARY_DIR}/package/native")
    # 宿主导出符号直连（TodStringTranslate 本地化翻译）：TaoEngineImport.cpp 的
    # 延迟加载钩子把 TaoEngineHost.dll 重定向到游戏 exe，分配桥统一跨模块堆。
    target_sources(pvz_a11y PRIVATE "${TAO_NATIVE_SDK}/runtime/TaoEngineImport.cpp")
    target_link_libraries(pvz_a11y PRIVATE "${TAO_NATIVE_SDK}/lib/TaoEngineHost.lib" delayimp sapi ole32)
    target_link_options(pvz_a11y PRIVATE "/DELAYLOAD:TaoEngineHost.dll" "/DELAY:NOBIND")
    set(TAO_GAME_EXE "${CMAKE_CURRENT_SOURCE_DIR}/../Tao-Windows-x64-20261005/Tao-Windows-x64-20261005/PlantsVsZombies.exe"
        CACHE PATH "游戏 exe 路径（engine_sdk 构建戳的宿主指纹来源）")
    add_custom_command(TARGET pvz_a11y POST_BUILD
        COMMAND "${CMAKE_COMMAND}" "-DHOST_EXE=${TAO_GAME_EXE}"
            "-DPLUGIN=$<TARGET_FILE:pvz_a11y>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/StampFromExe.cmake"
        VERBATIM)
elseif(ANDROID)
    include("${TAO_NATIVE_SDK}/cmake/TaoNativePlugin.cmake")
    tao_configure_native_plugin(pvz_a11y "${CMAKE_CURRENT_BINARY_DIR}/package/native")
    # Android uses shared libc++ and resolves host symbols from libmain.so
    if(EXISTS "${TAO_NATIVE_SDK}/platforms/android_arm64/lib/libmain.so")
        add_library(TaoEngineHost SHARED IMPORTED)
        set_target_properties(TaoEngineHost PROPERTIES IMPORTED_LOCATION "${TAO_NATIVE_SDK}/platforms/android_arm64/lib/libmain.so")
        target_link_libraries(pvz_a11y PRIVATE TaoEngineHost)
    elseif(EXISTS "${TAO_NATIVE_SDK}/lib/libmain.so")
        add_library(TaoEngineHost SHARED IMPORTED)
        set_target_properties(TaoEngineHost PROPERTIES IMPORTED_LOCATION "${TAO_NATIVE_SDK}/lib/libmain.so")
        target_link_libraries(pvz_a11y PRIVATE TaoEngineHost)
    endif()
    # Prism 跨平台静态库构建与链接 (Android TTS & Screen Reader 后端)
    set(PRISM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../prism")
    if(EXISTS "${PRISM_SOURCE_DIR}/CMakeLists.txt")
        set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static prism for mod" FORCE)
        set(PRISM_ENABLE_TESTS OFF CACHE BOOL "Disable tests" FORCE)
        set(PRISM_ENABLE_DEMOS OFF CACHE BOOL "Disable demos" FORCE)
        set(PRISM_ENABLE_GDEXTENSION OFF CACHE BOOL "Disable gdextension" FORCE)
        set(PRISM_ENABLE_SHIMS OFF CACHE BOOL "Disable shims" FORCE)
        add_subdirectory("${PRISM_SOURCE_DIR}" prism_native_build EXCLUDE_FROM_ALL)
        target_compile_definitions(pvz_a11y PRIVATE PVZ_A11Y_WITH_PRISM PRISM_STATIC)
        target_include_directories(pvz_a11y PRIVATE "${PRISM_SOURCE_DIR}/include")
        target_link_libraries(pvz_a11y PRIVATE prism)
    endif()
    target_link_libraries(pvz_a11y PRIVATE log)

    set_target_properties(pvz_a11y PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    target_link_options(pvz_a11y PRIVATE "-Wl,-z,max-page-size=16384")
    set(TAO_IDENTITY_JSON "${TAO_NATIVE_SDK}/platforms/android_arm64/identity.json")
    if(NOT EXISTS "${TAO_IDENTITY_JSON}")
        set(TAO_IDENTITY_JSON "${TAO_NATIVE_SDK}/identity.json")
    endif()
    if(EXISTS "${TAO_IDENTITY_JSON}")
        add_custom_command(TARGET pvz_a11y POST_BUILD
            COMMAND "${CMAKE_COMMAND}" "-DIDENTITY=${TAO_IDENTITY_JSON}" "-DPLUGIN=$<TARGET_FILE:pvz_a11y>"
                -P "${TAO_NATIVE_SDK}/cmake/StampPlugin.cmake" VERBATIM)
    endif()
endif()
configure_file(mod.lua "${CMAKE_CURRENT_BINARY_DIR}/package/mod.lua" COPYONLY)
configure_file(README.md "${CMAKE_CURRENT_BINARY_DIR}/package/README.md" COPYONLY)
configure_file(使用说明.md "${CMAKE_CURRENT_BINARY_DIR}/package/使用说明.md" COPYONLY)
configure_file(resources.xml "${CMAKE_CURRENT_BINARY_DIR}/package/resources.xml" COPYONLY)
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/sounds" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/package")

    if(WIN32 AND EXISTS "${PRISM_ROOT}/include/prism.h")
        add_executable(speech_tests tests/Speech.cpp)
